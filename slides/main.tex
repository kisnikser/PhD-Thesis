\begin{frame}{Масштабирование параметрических моделей машинного обучения на основе локальной геометрии оптимизационной поверхности}
    \begin{block}{Проблема}
        Эмпирические законы масштабирования воспроизводятся на практике, но не имеют общего теоретического объяснения. Множество оптимальных параметров зависит от конечной обучающей выборки.
    \end{block}
    \begin{block}{Требуется}
        Описать масштабирование параметрических моделей на основе локальной геометрии оптимизационной поверхности и определить по нему достаточный размер обучающей выборки.
    \end{block}
    \begin{block}{Метод решения}
        Анализ \alert{\textbf{стабилизации ландшафта функции потерь}} при росте выборки: для нейронных сетей "--- через спектр матрицы Гессе, для линейных моделей "--- через правдоподобие и апостериорные распределения.
    \end{block}
\end{frame}

\begin{frame}{Разложение Гаусса"--~Ньютона матрицы Гессе}
\begin{columns}[T]
    \begin{column}{0.56\textwidth}
        \begin{block}{Функция потерь и матрица Гессе}
            \vspace{-0.5em}
            \[
                \mathfrak{D}_m=\{(\mathbf{x}_i,\mathbf{y}_i)\}_{i=1}^m
            \]
            \[
                \mathbf{x}\in\mathbb{R}^{n},\quad
                \mathbf{y}\in\mathbb{R}^{K},\quad
                \mathbf{w}\in\mathbb{R}^{N}
            \]
            \[
                \mathcal{L}_m(\mathbf{w})=\frac1m\sum_{i=1}^m \ell_i(\mathbf{w}),
                \quad
                \ell_i(\mathbf{w})=\ell(f_{\mathbf{w}}(\mathbf{x}_i),\mathbf{y}_i)
            \]
            \[
                \mathbf{H}^{(m)}(\mathbf{w})
                =
                \nabla_{\mathbf{w}}^2\mathcal{L}_m(\mathbf{w})
                =
                \frac1m\sum_{i=1}^m\mathbf{H}_i(\mathbf{w})
            \]
            \vspace{-1.5em}
        \end{block}
        \begin{block}{Обозначения на одном объекте}
            \vspace{-1.4em}
            \[
                \mathbf{z}=f_{\mathbf{w}}(\mathbf{x}),
                \quad
                \mathbf{J}=\frac{\partial\mathbf{z}}{\partial\mathbf{w}},
                \quad
                \mathbf{A}=\nabla_{\mathbf{z}}^2\ell
            \]
            \vspace{-1.2em}
        \end{block}
    \end{column}
    \begin{column}{0.44\textwidth}
        \begin{block}{Разложение Гаусса"--~Ньютона}
            \vspace{-0.4em}
            \[
                \mathbf{H}(\mathbf{w})=\mathbf{G}(\mathbf{w})+\mathbf{R}(\mathbf{w})
            \]
            \[
                \mathbf{G}=\mathbf{J}^\top\mathbf{A}\mathbf{J}
            \]
            \[
                \mathbf{R}(\mathbf{w})
                =
                \sum_{c=1}^{K}
                \frac{\partial\ell}{\partial z_c}\,
                \nabla_{\mathbf{w}}^2 z_c
            \]
            \vspace{-1.1em}
        \end{block}
        \begin{block}{Ключевая идея}
            При малых невязках $\mathbf{R}(\mathbf{w})$ мало и $\|\mathbf{G}\|_2\approx\|\mathbf{H}\|_2$: кривизна анализируется через \alert{\textbf{компоненту $\mathbf{G}(\mathbf{w})$}}.
        \end{block}
    \end{column}
\end{columns}
\end{frame}

    \vspace{-0.4em}
    \begin{figure}
        \centering
        \resizebox{0.56\linewidth}{!}{\input{figs/gn_factorization.tikz}}

    \vspace{-0.8em}
    \begin{block}{Лемма (Киселев, 2026)}
        Сеть матрично-представима, если слой линеен по входу: $\mathbf{z}^{(p)}=T^{(p)}(\mathbf{x}^{(p)})$, $\mathbf{x}^{(p+1)}=\sigma^{(p)}(\mathbf{z}^{(p)})$. Тогда
        \[
            \mathbf{G}^{(p,q)}(\mathbf{w})
            =
            \mathbf{P}^{(p)\top}\mathbf{S}^{(p)\top}\mathbf{A}\,\mathbf{S}^{(q)}\mathbf{P}^{(q)},
        \]
        где $\mathbf{P}^{(p)}=\partial\mathbf{z}^{(p)}/\partial\mathbf{w}^{(p)}$ "--- якобиан слоя по своим параметрам, $\mathbf{S}^{(p)}=\partial\mathbf{z}/\partial\mathbf{z}^{(p)}$ "--- распространение возмущения до выхода.
    \end{block}
\end{frame}

\begin{frame}{Оценки спектральной нормы $\|\mathbf{G}\|_2$}
    Пусть $\|\mathbf{x}\|_2\leqslant M_{\mathbf{x}}$, $\|\mathbf{A}\|_2\leqslant M_{\mathbf{A}}$, $\|\mathbf{D}^{(p)}\|_2\leqslant M_{\sigma}$, где $\mathbf{D}^{(p)}$ "--- производные активации. Сети без смещений.
    \begin{block}{Теорема (Киселев, 2024)}
        $L$-слойная \alert{\textbf{полносвязная}} сеть, $\|\mathbf{W}^{(p)}\|_2\leqslant M_{\mathbf{W}}$:
        \[
            \|\mathbf{G}(\mathbf{w})\|_2
            \leqslant
            L M_{\mathbf{x}}^2 M_{\mathbf{A}}\,(M_{\mathbf{W}} M_{\sigma})^{2L-2}.
        \]
        \vspace{-1.2em}
    \end{block}
    \begin{block}{Теорема (Мешков, Киселев, 2024)}
        $L$-слойная \alert{\textbf{сверточная}} сеть, $\|\mathcal{W}^{(p)}\|_1\leqslant M_{\mathcal{W}}$, $K_{\max}=\max_p(C_{\mathrm{out}}^{(p)} k_h k_w)$:
        \[
            \|\mathbf{G}(\mathbf{w})\|_2
            \leqslant
            L M_{\mathbf{x}}^2 M_{\mathbf{A}}\,\alert{K_{\max}}\,(\alert{M_{\mathcal{W}}} M_{\sigma})^{2L-2}.
        \]
        \vspace{-1.2em}
    \end{block}
\end{frame}

\begin{frame}{Критерий сходимости поверхности функции потерь}
    К выборке $\mathfrak{D}_k$ с эмпирическим риском $\mathcal{L}_k$ и минимумом $\mathbf{w}_k^*$ добавляется один объект. Пробное распределение $q$ сосредоточено вблизи $\mathbf{w}_k^*$.
    \begin{block}{Определение}
        \vspace{-0.8em}
        \[
            \Delta_p(k+1)
            =
            \int\bigl|\mathcal{L}_{k+1}(\mathbf{w})-\mathcal{L}_k(\mathbf{w})\bigr|^p q(\mathbf{w})\,d\mathbf{w}.
        \]
        \vspace{-1.3em}
    \end{block}
    \begin{block}{Утверждение (точное представление)}
        \vspace{-0.8em}
        \[
            \mathcal{L}_{k+1}(\mathbf{w})-\mathcal{L}_k(\mathbf{w})
            =
            \frac{\ell_{k+1}(\mathbf{w})-\mathcal{L}_k(\mathbf{w})}{k+1},
        \]
        поэтому $\Delta_p(k+1)=\mathbb{E}_q|\ell_{k+1}-\mathcal{L}_k|^p/(k+1)^p$.
        Если $|\ell_i|\leqslant M_{\ell}$, то $\Delta_p(k+1)\leqslant(2M_{\ell})^p/(k+1)^p$.
        \vspace{-0.6em}

        Порядок $k^{-p}$ возникает до разложения Тейлора. Квадратичная аппроксимация и матрица Гессе оценивают числитель, а не порядок.
    \end{block}
\end{frame}

\begin{frame}{Абсолютный одноточечный критерий сходимости}
    \begin{columns}[T]
        \begin{column}{0.58\textwidth}
            Одноточечный критерий "--- изменение ландшафта в одной точке окрестности минимума радиуса $R$:
            \[
                \Delta_1(k+1)=\bigl|\mathcal{L}_{k+1}(\mathbf{w})-\mathcal{L}_k(\mathbf{w})\bigr|,
            \]
            то есть $p=1$ и $q$ "--- дельта-мера в точке $\mathbf{w}\in\mathcal{U}_R(\mathbf{w}_k^*)$.
        \end{column}
        \begin{column}{0.40\textwidth}
            \centering
            \includegraphics[width=\linewidth]{criterion_abs.pdf}
        \end{column}
    \end{columns}
    \begin{block}{Теорема (Киселев, 2024)}
        В $\mathcal{U}_R(\mathbf{w}_k^*)$ верна квадратичная аппроксимация, $\|\mathbf{g}_{k+1}(\mathbf{w}_k^*)\|_2\leqslant M_{\mathbf{g}}$, $|\ell_i(\mathbf{w}_k^*)|\leqslant M_{\ell}$ и $\|\mathbf{H}_i(\mathbf{w}_k^*)\|_2\leqslant M_{\mathbf{H}}$. Тогда
        \[
            \Delta_1(k+1)
            \leqslant
            \frac{2M_{\ell}+M_{\mathbf{g}}R+M_{\mathbf{H}}R^2}{k+1}
            =
            \mathcal{O}(k^{-1}).
        \]
        \vspace{-1.2em}
    \end{block}
\end{frame}

\begin{frame}{Среднеквадратичный критерий сходимости}
    \begin{columns}[T]
        \begin{column}{0.58\textwidth}
            Средний квадрат того же приращения по гауссовскому пробному распределению $q=\mathcal{N}(\mathbf{w}_k^*,\sigma^2\mathbf{I}_N)$:
            \[
                \Delta_2(k+1)
                =
                \mathbb{E}_q\bigl(\mathcal{L}_{k+1}(\mathbf{w})-\mathcal{L}_k(\mathbf{w})\bigr)^2.
            \]
            На практике среднее заменяется выборкой Монте-Карло $\mathbf{w}_s\sim q$.
        \end{column}
        \begin{column}{0.40\textwidth}
            \centering
            \includegraphics[width=0.80\linewidth]{criterion_squared.pdf}
        \end{column}
    \end{columns}
    \begin{block}{Теорема (Киселев, 2025)}
        В предположениях предыдущей теоремы
        \[
            \Delta_2(k+1)
            \leqslant
            \frac{12M_{\ell}^2+3\sigma^2 M_{\mathbf{g}}^2+3\sigma^4(\alert{N^2+2N})M_{\mathbf{H}}^2}{(k+1)^2}
            =
            \mathcal{O}(k^{-2}).
        \]
        \vspace{-1.1em}

        Множитель $N^2+2N$ растет с числом параметров, и константа теряет практический смысл.
    \end{block}
\end{frame}

\begin{frame}{Критерий в подпространстве наибольшей кривизны}
    \begin{columns}[T]
        \begin{column}{0.58\textwidth}
            Пусть $\lambda_1^{(k)}\geqslant\ldots\geqslant\lambda_N^{(k)}$ "--- собственные значения $\mathbf{H}^{(k)}(\mathbf{w}_k^*)$, $\mathbf{U}_D=[\mathbf{u}_1,\ldots,\mathbf{u}_D]$, $\mathcal{S}_D=\mathrm{Im}\,\mathbf{U}_D$. Пробное распределение сосредоточено в $\mathbf{w}_k^*+\mathcal{S}_D$:
            \[
                \mathbf{w}=\mathbf{w}_k^*+\mathbf{U}_D\mathbf{z},
                \quad
                \mathbf{z}\sim\mathcal{N}(\mathbf{0},\sigma^2\mathbf{I}_D).
            \]
        \end{column}
        \begin{column}{0.40\textwidth}
            \centering
            \includegraphics[width=\linewidth]{criterion_squared_subspace.pdf}
        \end{column}
    \end{columns}
    \begin{block}{Теорема (Киселев, 2026)}
        В предположениях предыдущих теорем
        \[
            \Delta_2^{(D)}(k+1)
            \leqslant
            \frac{12M_{\ell}^2+3\sigma^2 M_{\mathbf{g}}^2+3\sigma^4(\alert{D^2+2D})M_{\mathbf{H}}^2}{(k+1)^2}
            =
            \mathcal{O}(k^{-2}).
        \]
        \vspace{-1.2em}

        Порядок тот же. В константе $N$ заменено размерностью подпространства $D$.
    \end{block}
\end{frame}

\begin{frame}{Достаточный размер выборки}
    Размер $m^*$ называется $\varepsilon$-достаточным по критерию $\Delta$, если $\Delta(m)\leqslant\varepsilon$ для всех $m\geqslant m^*$. Подстановка $\|\mathbf{H}_i\|_2\leqslant M_{\mathbf{G}}(\mathcal{A})+\delta_R$ в оценки предыдущих теорем дает явные пороги.
    \begin{block}{Теорема (Киселев, 2026). Одноточечный критерий}
        \vspace{-0.9em}
        \[
            m_1^*(\varepsilon,\mathcal{A})
            =
            \frac{2M_{\ell}+M_{\mathbf{g}}R+\bigl(M_{\mathbf{G}}(\mathcal{A})+\delta_R\bigr)R^2}{\varepsilon}.
        \]
        \vspace{-1.3em}
    \end{block}
    \begin{block}{Теорема (Киселев, 2026). Среднеквадратичный критерий}
        \vspace{-0.9em}
        \[
            m_2^*(\varepsilon,\mathcal{A})
            =
            \sqrt{\frac{12M_{\ell}^2+3\sigma^2 M_{\mathbf{g}}^2+3\sigma^4(N^2+2N)\bigl(M_{\mathbf{G}}(\mathcal{A})+\delta_R\bigr)^2}{\varepsilon}}.
        \]
        \vspace{-1.3em}
    \end{block}
    \begin{block}{Теорема (Киселев, 2026). Подпространство размерности $D$}
        \vspace{-0.9em}
        \[
            m_3^*(\varepsilon,\mathcal{A},D)
            =
            \sqrt{\frac{12M_{\ell}^2+3\sigma^2 M_{\mathbf{g}}^2+3\sigma^4(D^2+2D)\bigl(M_{\mathbf{G}}(\mathcal{A})+\delta_R\bigr)^2}{\varepsilon}}.
        \]
        \vspace{-1.3em}
    \end{block}
\end{frame}

\begin{frame}{Схема закона масштабирования}
    Средний риск в точке решения раскладывается тождественно:
    \[
        \mathbb{E}\,\mathcal{L}(\mathbf{w}_m^*)=E+\Phi(N)+\mathcal{E}(m),
    \]
    где $E$ "--- неустранимая ошибка, $\Phi(N)$ "--- ошибка выразимости класса с $N$ параметрами, $\mathcal{E}(m)\geqslant 0$ "--- ошибка оценивания.
    \begin{block}{Предположение (Киселев, 2026). Локальный статистический режим}
        Существуют $\rho\in(0,1]$ и $C>0$ такие, что $\mathcal{E}(m)\leqslant C\bigl(M_{\mathbf{G}}(\mathcal{A})+\delta_R\bigr)\,m^{-\rho}$. Это структурное предположение: оно не следует из предыдущих глав.
    \end{block}
    \begin{block}{Предложение (Киселев, 2026)}
        \vspace{-0.8em}
        \[
            \mathbb{E}\,\mathcal{L}(\mathbf{w}_m^*)
            \leqslant
            E+\Phi(N)+\frac{C\bigl(M_{\mathbf{G}}(\mathcal{A})+\delta_R\bigr)}{m^{\rho}}.
        \]
        \vspace{-1.2em}

        Для линейной регрессии предположение выполнено точно с $\rho=1$, а константа ошибки оценивания есть $\operatorname{tr}(\mathbf{H}^{-1}\mathbf{C})/2$, не $\|\mathbf{G}\|_2$.
    \end{block}
\end{frame}

\begin{frame}{Линейная модель с неполным набором признаков}
    Признаки $\mathbf{x}\sim\mathcal{N}(\mathbf{0},\boldsymbol{\Lambda})$, $\boldsymbol{\Lambda}=\operatorname{diag}(\lambda_j)$, ответ $y=\sum_j\theta_j^\star x_j+\varepsilon$. Модель с $N$ параметрами видит первые $N$ признаков. $e_j=\lambda_j(\theta_j^\star)^2$, $\Phi(N)=\frac12\sum_{j>N}e_j$, $\sigma_N^2=\sigma^2+2\Phi(N)$, $E=\sigma^2/2$.
    \begin{block}{Теорема (Киселев, 2026)}
        При $m>N+1$
        \[
            \mathbb{E}\,\mathcal{L}(\hat{\mathbf{w}}_{N,m})
            =
            \frac{\sigma^2}{2}+\Phi(N)+\frac{\sigma_N^2 N}{2(m-N-1)},
        \]
        \[
            \mathcal{E}(N,m)
            =
            \frac{\operatorname{tr}(\mathbf{H}_N^{-1}\mathbf{C}_N)}{2m}
            \left(1+\frac{N+1}{m-N-1}\right),
            \quad
            \mathbf{H}_N=\boldsymbol{\Lambda}_N,\ 
            \mathbf{C}_N=\sigma_N^2\boldsymbol{\Lambda}_N.
        \]
        \vspace{-0.8em}
    \end{block}
    \begin{block}{Следствие (Киселев, 2026). При $e_j=c\,j^{-b}$, $b>1$}
        $\alpha=b-1$, при фиксированном $N$ верно $\rho=1$, а при бюджете $\mathcal{C}=Nm$ вычислительно-оптимальный размер $N^*\asymp\mathcal{C}^{1/(\alpha+2)}$.
    \end{block}
\end{frame}

    \end{block}
\end{frame}

\begin{frame}{Критерии достаточного размера для линейных моделей}
    Подвыборки $\mathfrak{D}_k\subset\mathfrak{D}_m$, $\hat{\mathbf{w}}_k$ "--- оценка максимума правдоподобия, $p_k$ "--- апостериор. Схожие подвыборки отличаются одним объектом.
    \begin{block}{Теорема (Киселев, 2024). $M$-достаточность}
        $M(k+1)=\bigl|\mathbb{E} L(\mathfrak{D}_m,\hat{\mathbf{w}}_{k+1})-\mathbb{E} L(\mathfrak{D}_m,\hat{\mathbf{w}}_k)\bigr|$. Если средние и ковариации $\hat{\mathbf{w}}_k$ сходятся, то для любого $\varepsilon>0$ найдется $m^*$ с $M(k)\leqslant\varepsilon$ при $k\geqslant m^*$.
    \end{block}
    \begin{block}{Теорема (Киселев, 2024). $KL$- и $S$-достаточность}
        $\mathrm{KL}(k+1)=D_{\mathrm{KL}}(p_k\|p_{k+1})$, $S$ "--- s-score тех же апостериоров. Для нормальных апостериоров сходимость средних и ковариаций дает $\mathrm{KL}(k)\leqslant\varepsilon$; для $S(k)\geqslant 1-\varepsilon$ достаточно сходимости средних.
    \end{block}
    \begin{block}{Теорема (Киселев, 2025). Моменты}
        При $\|\mathbf{x}\|_2\leqslant M$, $|y|\leqslant M$ и $\lambda_{\min}(\mathbf{X}_k^\top\mathbf{X}_k)=\omega(\sqrt{k})$ средние и ковариации апостериора линейной регрессии сходятся, так что условия двух предыдущих теорем выполнены.
    \end{block}
\end{frame}

    \end{block}
\end{frame}

        \item Критерии достаточного размера выборки для вероятностных линейных моделей, основанные на стабилизации распределения оптимальных параметров при увеличении объема данных.
    \end{enumerate}
\end{frame}
