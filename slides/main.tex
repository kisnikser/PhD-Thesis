\begin{frame}{Ландшафтный метод оценки масштабируемости параметрических моделей машинного обучения}
    \begin{block}{Проблема}
        Эмпирические законы масштабирования воспроизводятся на практике, но не имеют общего теоретического объяснения. Множество оптимальных параметров зависит от конечной обучающей выборки.
    \end{block}
    \begin{block}{Требуется}
        Описать зависимость качества параметрической модели заданной архитектуры от ее размера и~объема обучающих данных для их совместного выбора и~определить достаточный размер выборки.
    \end{block}
    \begin{block}{Метод решения}
        Анализ локальной геометрии \alert{\textbf{ландшафта функции потерь}} и~ее стабилизации при росте выборки: для нейронных сетей "--- через спектр матрицы Гессе, для линейных моделей "--- через правдоподобие и~апостериорные распределения.
    \end{block}
\end{frame}

\begin{frame}{Постановка задачи\chap{1}}
    \small
    \begin{block}{Выборка и~модель}
        Объект $(\mathbf x,\mathbf y)$: $\mathbf x\in\mathbb R^n$ "--- признаковое описание, $\mathbf y\in\mathbb R^K$ "--- целевая переменная. Выборка $\mathfrak D_m=\{(\mathbf x_i,\mathbf y_i)\}_{i=1}^m$ состоит из~$m$ независимых реализаций пары $(\mathbf x,\mathbf y)$ из неизвестного распределения $p(\mathbf x,\mathbf y)$. Параметрическая модель $f_{\mathbf w}:\mathbb R^n\to\mathbb R^K$, $\mathbf w\in\mathbb R^N$, аппроксимирует зависимость $\mathbf y$ от~$\mathbf x$.
    \end{block}
    \begin{block}{Задача обучения}
        \vspace{-1.0em}
        \[
            \mathcal L_m(\mathbf w)=\frac1m\sum_{i=1}^m\underbrace{\ell\bigl(f_{\mathbf w}(\mathbf x_i),\mathbf y_i\bigr)}_{\ell_i(\mathbf w)}\ \to\ \min_{\mathbf w\in\mathbb R^N},
            \qquad
            \mathbf w_m^*\in\argmin_{\mathbf w\in\mathbb R^N}\mathcal L_m(\mathbf w),
        \]
        \[
            \mathcal L(\mathbf w)=\mathbb E_{(\mathbf x,\mathbf y)\sim p}\,\ell\bigl(f_{\mathbf w}(\mathbf x),\mathbf y\bigr)
            \quad\text{"--- популяционный риск}.
        \]
        \vspace{-1.3em}
    \end{block}
    \begin{block}{Локальная геометрия ландшафта}
        \vspace{-1.0em}
        \[
            \mathbf g^{(m)}(\mathbf w)=\nabla_{\mathbf w}\mathcal L_m(\mathbf w)\in\mathbb R^N,
            \qquad
            \mathbf H^{(m)}(\mathbf w)=\nabla^2_{\mathbf w}\mathcal L_m(\mathbf w)=\frac1m\sum_{i=1}^m\mathbf H_i(\mathbf w)\in\mathbb R^{N\times N}.
        \]
        \vspace{-1.3em}
    \end{block}
\end{frame}

%======================================================================
\begin{frame}{Разложение Гаусса"--~Ньютона\chap{1}}
    \begin{block}{Обозначения на одном объекте $(\mathbf x,\mathbf y)$}
        \vspace{-1.0em}
        \[
            \mathbf z=f_{\mathbf w}(\mathbf x)\in\mathbb R^{K},
            \qquad
            \mathbf J(\mathbf w)=\frac{\partial\mathbf z}{\partial\mathbf w}\in\mathbb R^{K\times N},
            \qquad
            \mathbf A(\mathbf w)=\nabla_{\mathbf z}^2\ell(\mathbf z,\mathbf y)\in\mathbb R^{K\times K}.
        \]
        \vspace{-1.3em}
    \end{block}
    \begin{block}{Разложение матрицы Гессе на одном объекте}
        \vspace{-1.0em}
        \[
            \mathbf H(\mathbf w)=\mathbf G(\mathbf w)+\mathbf R(\mathbf w),
            \qquad
            \mathbf G(\mathbf w)=\mathbf J(\mathbf w)^\top\mathbf A(\mathbf w)\mathbf J(\mathbf w),
            \qquad
            \mathbf R(\mathbf w)=\sum_{c=1}^{K}\frac{\partial\ell(\mathbf z,\mathbf y)}{\partial z_c}\,\nabla_{\mathbf w}^2 z_c.
        \]
        \vspace{-1.3em}
    \end{block}
    \begin{block}{Ключевая идея}
        Остаточная компонента $\mathbf R(\mathbf w)$ пропорциональна производным $\partial\ell/\partial z_c$ и~\alert{\textbf{ослабевает в~окрестности минимума}}, а~также вносит вклад преимущественно в~область малых собственных значений. Спектрально значимую часть матрицы Гессе задает \alert{\textbf{гаусс"--~ньютоновская компонента $\mathbf G(\mathbf w)$}}: через нее в~кривизну входит архитектура модели.
    \end{block}
\end{frame}

%======================================================================
\begin{frame}{Блочная факторизация гаусс"--~ньютоновской компоненты\chap{1}}
    \small
    \begin{block}{Определение}
        Сеть \alert{\textbf{матрично-представима}}, если $\mathbf x^{(1)}=\mathbf x$, $\mathbf z^{(p)}=T^{(p)}_{\mathbf w^{(p)}}(\mathbf x^{(p)})$, $\mathbf x^{(p+1)}=\sigma^{(p)}(\mathbf z^{(p)})$, $f_{\mathbf w}(\mathbf x)=\mathbf z^{(L)}$, где оператор $T^{(p)}_{\mathbf w^{(p)}}$ линеен по входу и~допускает матричное представление по~параметрам $\mathbf w^{(p)}\in\mathbb R^{N_p}$.
    \end{block}
    \vspace{-0.6em}
    \begin{figure}
        \centering
        \resizebox{0.47\linewidth}{!}{\input{figs/gn_factorization.tikz}}
    \end{figure}
    \vspace{-1.0em}
    \begin{block}{Лемма (Киселев, 2026)}
        Пусть $f_{\mathbf w}$ "--- матрично-представимая сеть. Тогда $\mathbf G(\mathbf w)=\bigl[\mathbf G^{(p,q)}(\mathbf w)\bigr]_{p,q=1}^L$, где
        \[
            \mathbf G^{(p,q)}(\mathbf w)=\mathbf P^{(p)\top}\mathbf S^{(p)\top}\mathbf A\,\mathbf S^{(q)}\mathbf P^{(q)},
            \qquad
            \mathbf P^{(p)}=\frac{\partial\mathbf z^{(p)}}{\partial\mathbf w^{(p)}},
            \quad
            \mathbf S^{(p)}=\frac{\partial\mathbf z}{\partial\mathbf z^{(p)}}.
        \]
        \vspace{-1.3em}
    \end{block}
\end{frame}

\begin{frame}{Архитектурно-зависимые оценки $\|\mathbf G\|_2$\chap{1}}
    \small
    \begin{block}{Теорема (Киселев, 2024)}
        Пусть $f_{\mathbf w}$ "--- $L$-слойная \alert{\textbf{полносвязная}} сеть без смещений и~для всех $p=1,\ldots,L$
        \[
            \|\mathbf x\|_2\leqslant M_{\mathbf x},
            \quad
            \|\mathbf A(\mathbf w)\|_2\leqslant M_{\mathbf A},
            \quad
            \|\mathbf W^{(p)}\|_2\leqslant M_{\mathbf W},
            \quad
            \|\mathbf D^{(p)}\|_2\leqslant M_{\sigma},
        \]
        где $\mathbf D^{(p)}$ "--- матрица производных активации. Тогда
        \[
            \|\mathbf G(\mathbf w)\|_2\leqslant L M_{\mathbf x}^2 M_{\mathbf A}\,(M_{\mathbf W}M_{\sigma})^{2L-2}.
        \]
        Если, кроме того, $\sigma=\mathrm{ReLU}$, а~$\ell$ "--- кросс-энтропия, то $\|\mathbf G(\mathbf w)\|_2\leqslant\sqrt2\,L M_{\mathbf x}^2 M_{\mathbf W}^{2L-2}$.
    \end{block}
    \begin{block}{Теорема (Мешков, Киселев, 2024)}
        Пусть $f_{\mathbf w}$ "--- $L$-слойная \alert{\textbf{сверточная}} сеть без смещений, для всех $p=1,\ldots,L$ выполнено $\|\mathbf x\|_2\leqslant M_{\mathbf x}$, $\|\mathbf A(\mathbf w)\|_2\leqslant M_{\mathbf A}$, $\|\mathcal W^{(p)}\|_1\leqslant M_{\mathcal W}$, $\|\mathbf D^{(p)}\|_2\leqslant M_{\sigma}$ и~$K_{\max}=\max_p C_{\mathrm{out}}^{(p)}k_hk_w$. Тогда
        \[
            \|\mathbf G(\mathbf w)\|_2\leqslant L M_{\mathbf x}^2 M_{\mathbf A}\,K_{\max}\,(M_{\mathcal W}M_{\sigma})^{2L-2}.
        \]
        \vspace{-1.3em}
    \end{block}
\end{frame}

\begin{frame}{Эксперимент: гаусс"--~ньютоновская компонента и~матрица Гессе\chap{1}}
    \small
    \textbf{Цель:} проверить, что $\|\mathbf G\|_2$ близка к~$\|\mathbf H\|_2$, и~сравнить верхние оценки с~измеренными нормами.

Нормы вычисляются степенным методом на мини-пакете $128$ после инициализации и~эпох $1$, $5$, $10$.

\begin{frame}{Критерий сходимости поверхности функции потерь\chap{2}}
\small
\begin{columns}[T]
    \begin{column}{0.57\textwidth}
        \begin{minipage}{\linewidth}\raggedright
            Критерий $p$-сходимости ландшафта функции потерь при добавлении одного объекта в~выборку $\mathfrak D_k$ с~минимумом $\mathbf w_k^*$ функции $\mathcal L_k$:
            \[
                \Delta_p(k+1)=\int\bigl|\mathcal L_{k+1}(\mathbf w)-\mathcal L_k(\mathbf w)\bigr|^p q(\mathbf w)\,d\mathbf w,
            \]
            где плотность $q$ с~основной массой вблизи $\mathbf w_k^*$ задает \alert{\textbf{функцию предпочтения}} точек в~пространстве параметров.
        \end{minipage}
    \end{column}
    \begin{column}{0.40\textwidth}
        \centering
        \includegraphics[width=\linewidth]{criterion_general.pdf}
    \end{column}
\end{columns}
\begin{block}{Утверждение (тождество усреднения)}
    Для любой точки $\mathbf w$
    \[
        \mathcal L_{k+1}(\mathbf w)-\mathcal L_k(\mathbf w)=\frac{\ell_{k+1}(\mathbf w)-\mathcal L_k(\mathbf w)}{k+1}.
    \]
    Поэтому если $|\ell_i(\mathbf w)|\leqslant M_\ell$ для всех $i$, то $\Delta_p(k+1)\leqslant(2M_\ell)^p/(k+1)^p$: порядок $k^{-p}$ задается усреднением по~объектам, а~локальная геометрия ландшафта определяет константу.
\end{block}
\end{frame}

\begin{frame}{Схема теоретического анализа критериев\chap{2}}
\footnotesize
\begin{columns}[T]
    \begin{column}{0.64\textwidth}
        \begin{block}{Шаг 1. Квадратичная аппроксимация в~$\mathcal U_R(\mathbf w_k^*)$}
            С~учетом $\mathbf g^{(k)}(\mathbf w_k^*)=\mathbf 0$ и~$\mathbf d=\mathbf w-\mathbf w_k^*$:
            \[
                \mathcal L_{k+1}(\mathbf w)-\mathcal L_k(\mathbf w)=a_k+\mathbf b_k^\top\mathbf d+\tfrac12\mathbf d^\top\mathbf A_k\mathbf d,
            \]
            \[
                a_k=\mathcal L_{k+1}(\mathbf w_k^*)-\mathcal L_k(\mathbf w_k^*),
                \qquad
                \mathbf b_k=\mathbf g^{(k+1)}(\mathbf w_k^*)=\frac{\mathbf g_{k+1}(\mathbf w_k^*)}{k+1},
            \]
            \[
                \mathbf A_k=\mathbf H^{(k+1)}(\mathbf w_k^*)-\mathbf H^{(k)}(\mathbf w_k^*).
            \]
            \vspace{-1.3em}
        \end{block}
        \begin{block}{Шаг 2. Слабое градиентное рассогласование}
            $\|\mathbf g_{k+1}(\mathbf w_k^*)\|_2\leqslant M_{\mathbf g}$, т.\,е. $\|\nabla\mathcal L_{k+1}(\mathbf w_k^*)\|_2\leqslant M_{\mathbf g}/(k+1)$: новый объект почти не нарушает стационарность точки $\mathbf w_k^*$.
        \end{block}
        \begin{block}{Шаг 3. Оценка слагаемых и~интегрирование по~$q$}
            При $|\ell_i(\mathbf w_k^*)|\leqslant M_\ell$ и~$\|\mathbf H_i(\mathbf w_k^*)\|_2\leqslant M_{\mathbf H}$
            \[
                |a_k|\leqslant\frac{2M_\ell}{k+1},
                \qquad
                \|\mathbf b_k\|_2\leqslant\frac{M_{\mathbf g}}{k+1},
                \qquad
                \|\mathbf A_k\|_2\leqslant\frac{2M_{\mathbf H}}{k+1}.
            \]
            \vspace{-1.3em}
        \end{block}
    \end{column}
    \begin{column}{0.35\textwidth}
        \centering
        \includegraphics[width=\linewidth,height=0.84\textheight,keepaspectratio]{losses_assumption.pdf}
    \end{column}
\end{columns}
\end{frame}

\begin{frame}{Абсолютный одноточечный критерий\chap{2}}
\begin{columns}[T]
    \begin{column}{0.57\textwidth}
        \begin{minipage}{\linewidth}\raggedright
            Абсолютный одноточечный критерий сходимости:
            \[
                \Delta_1(k+1)=\bigl|\mathcal L_{k+1}(\mathbf w)-\mathcal L_k(\mathbf w)\bigr|,
            \]
            где $\mathbf w\in\mathcal U_R(\mathbf w_k^*)$ "--- точка из окрестности радиуса $R>0$ минимума $\mathbf w_k^*$, т.\,е. $p=1$ и~$q(\tilde{\mathbf w})=\delta(\tilde{\mathbf w}-\mathbf w)$.
        \end{minipage}
    \end{column}
    \begin{column}{0.40\textwidth}
        \centering
        \includegraphics[width=\linewidth]{criterion_abs.pdf}
    \end{column}
\end{columns}
\begin{block}{Теорема (Киселев, 2024)}
    Пусть в~$\mathcal U_R(\mathbf w_k^*)$ справедлива квадратичная аппроксимация, $\|\mathbf g_{k+1}(\mathbf w_k^*)\|_2\leqslant M_{\mathbf g}$ и~для всех $i=1,\ldots,k+1$ верно $|\ell_i(\mathbf w_k^*)|\leqslant M_\ell$, $\|\mathbf H_i(\mathbf w_k^*)\|_2\leqslant M_{\mathbf H}$. Тогда
    \[
        \Delta_1(k+1)\leqslant\frac{2M_\ell+M_{\mathbf g}R+M_{\mathbf H}R^2}{k+1}=\mathcal O(k^{-1}).
    \]
    \vspace{-1.2em}
\end{block}
\end{frame}

\begin{frame}{Среднеквадратичный критерий\chap{2}}
\begin{columns}[T]
    \begin{column}{0.57\textwidth}
        \begin{minipage}{\linewidth}\raggedright
            Среднеквадратичный критерий сходимости:
            \[
                \Delta_2(k+1)=\mathbb E_{q(\mathbf w)}\Bigl[\bigl(\mathcal L_{k+1}(\mathbf w)-\mathcal L_k(\mathbf w)\bigr)^2\Bigr],
            \]
            т.\,е. $p=2$ и~$q(\mathbf w)=\mathcal N(\mathbf w_k^*,\sigma^2\mathbf I_N)$; на практике критерий оценивается методом Монте"--~Карло по точкам $\mathbf w_s\sim q(\mathbf w)$.
        \end{minipage}
    \end{column}
    \begin{column}{0.40\textwidth}
        \centering
        \includegraphics[width=\linewidth]{criterion_squared.pdf}
    \end{column}
\end{columns}
\begin{block}{Теорема (Киселев, 2025)}
    Пусть выполнены условия предыдущей теоремы. Тогда
    \[
        \Delta_2(k+1)\leqslant\frac{12M_\ell^2+3\sigma^2M_{\mathbf g}^2+3\sigma^4(\alert{N^2+2N})M_{\mathbf H}^2}{(k+1)^2}=\mathcal O(k^{-2}).
    \]
    \vspace{-1.2em}
\end{block}
\end{frame}

\begin{frame}{Критерий в~подпространстве наибольшей кривизны\chap{2}}

        \Delta_2^{(D)}(k+1)\leqslant\frac{12M_\ell^2+3\sigma^2M_{\mathbf g}^2+3\sigma^4(\alert{D^2+2D})M_{\mathbf H}^2}{(k+1)^2}=\mathcal O(k^{-2}).
    \]
    \vspace{-1.2em}
\end{block}
\end{frame}

\begin{frame}{Эксперимент: геометрия ландшафта вдоль главных направлений\chap{2}}

\begin{frame}{Явные оценки достаточного размера выборки\chap{3}}

            m_1^*(\varepsilon,\mathcal A)=\frac{2M_\ell+M_{\mathbf g}R+\bigl(M_{\mathbf G}(\mathcal A)+\delta_R\bigr)R^2}{\varepsilon}

            m_2^*(\varepsilon,\mathcal A)=\sqrt{\frac{C(N)}{\varepsilon}},

\begin{frame}{Схема закона масштабирования\chap{3}}
    \small
    Популяционный риск $\mathcal L(\mathbf w)=\mathbb E_{(\mathbf x,\mathbf y)\sim p}\,\ell(f_{\mathbf w}(\mathbf x),\mathbf y)$; $\mathcal W_N\subset\mathbb R^N$ "--- модели с~$N$ параметрами, $\mathbf w_\infty^*$ "--- минимизатор риска в~предельном классе, $\mathbf w_m^*$ "--- решение по~выборке $\mathfrak D_m$.
    \begin{block}{Декомпозиция (тождество)}
        \vspace{-1.0em}
        \[
            \mathbb E_{\mathfrak D_m}\mathcal L(\mathbf w_m^*)
            =\underbrace{\mathcal L(\mathbf w_\infty^*)}_{E\text{ "--- неустранимая}}
            +\underbrace{\inf_{\mathbf w\in\mathcal W_N}\mathcal L(\mathbf w)-E}_{\Phi(N)\text{ "--- выразимость}}
            +\underbrace{\mathbb E_{\mathfrak D_m}\mathcal L(\mathbf w_m^*)-\inf_{\mathbf w\in\mathcal W_N}\mathcal L(\mathbf w)}_{\mathcal E(m)\geqslant0\text{ "--- оценивание}}.
        \]
        \vspace{-1.2em}
    \end{block}
    \begin{block}{Предположение (локальный статистический режим)}
        Пусть $\mathbf w_m^*$ лежит в~области, где квадратичная аппроксимация информативна, $\delta_R$ мало по~сравнению с~$M_{\mathbf G}(\mathcal A)$, а~минимум устойчив. Тогда существуют $\rho\in(0,1]$ и~$C>0$ такие, что $\mathcal E(m)\leqslant C\bigl(M_{\mathbf G}(\mathcal A)+\delta_R\bigr)m^{-\rho}$.
    \end{block}
    \begin{block}{Утверждение (Киселев, 2026)}
        Пусть выполнено предположение о~локальном режиме. Тогда
        \[
            \mathbb E_{\mathfrak D_m}\mathcal L(\mathbf w_m^*)\leqslant E+\Phi(N)+\frac{C\bigl(M_{\mathbf G}(\mathcal A)+\delta_R\bigr)}{m^{\rho}}.
        \]
        \vspace{-1.3em}
    \end{block}
\end{frame}

\begin{frame}{Точно решаемый случай: линейная регрессия\chap{3}}
    \small
    \begin{block}{Утверждение (Киселев, 2026)}
        Пусть $y=\mathbf w_\star^\top\mathbf x+\varepsilon$, где $\mathbf x\sim\mathcal N(\mathbf 0,\boldsymbol\Sigma)$, $\boldsymbol\Sigma\succ0$, шум $\varepsilon\sim\mathcal N(0,\sigma^2)$ не зависит от $\mathbf x$, $\ell=\frac12(y-\mathbf w^\top\mathbf x)^2$, а~$\hat{\mathbf w}_m$ "--- оценка наименьших квадратов по~$m>n+1$ объектам. Тогда
        \[
            \mathcal E(m)=\mathbb E_{\mathfrak D_m}\mathcal L(\hat{\mathbf w}_m)-\mathcal L(\mathbf w_\star)
            =\frac{\sigma^2n}{2(m-n-1)}
            =\frac{\tr(\mathbf H^{-1}\mathbf C)}{2m}\Bigl(1+\frac{n+1}{m-n-1}\Bigr),
        \]
        где $\mathbf H=\boldsymbol\Sigma$ "--- матрица Гессе, $\mathbf C=\operatorname{Cov}\nabla\ell(\mathbf w_\star)=\sigma^2\boldsymbol\Sigma$ "--- ковариация градиентов.
    \end{block}
    \begin{block}{Лемма}
        Пусть $\mathbf H\succ0$, $\mathbf C\succeq0$, $\nabla\ell_i=\mathbf J_i^\top\mathbf s_i$, $\mathbf A_i\succeq a_{\min}\mathbf I$, $a_{\min}>0$, и~$\|\mathbf G_i(\mathbf w^*)\|_2\leqslant M_{\mathbf G}(\mathcal A)$. Тогда
        \[
            \tr(\mathbf H^{-1}\mathbf C)\leqslant\|\mathbf C\|_2\tr(\mathbf H^{-1})\leqslant\frac{\mathbb E\|\mathbf s\|^2}{a_{\min}}\,M_{\mathbf G}(\mathcal A)\tr(\mathbf H^{-1}).
        \]
        \vspace{-1.3em}
    \end{block}
    \begin{block}{Следствие}
        Пусть, кроме того, $\|\mathbf x\|\leqslant M_{\mathbf x}$ и~$m\geqslant2(n+1)$. Тогда предположение о~локальном режиме выполнено \alert{\textbf{точно}}: $\rho=1$, $\delta_R=0$, $M_{\mathbf G}=M_{\mathbf x}^2$, $C=\sigma^2\tr(\boldsymbol\Sigma^{-1})$.
    \end{block}
\end{frame}

\begin{frame}{Вероятностная модель линейной регрессии\chap{4}}
    \small
    \begin{block}{Постановка}
        Объект $(\mathbf x,y)$: $\mathbf x\in\mathbb R^n$, $y\in\mathbb R$; выборка $\mathfrak D_m$ из~$m$ независимых реализаций, $\mathbf X_m=[\mathbf x_1,\ldots,\mathbf x_m]^\top\in\mathbb R^{m\times n}$, $\mathbf y_m\in\mathbb R^m$. Модель задается правдоподобием
        \[
            p(y\,|\,\mathbf x,\mathbf w)=\mathcal N(y\,|\,\mathbf w^\top\mathbf x,\sigma^2),
            \qquad
            L(\mathfrak D_m,\mathbf w)=p(\mathbf y_m\,|\,\mathbf X_m,\mathbf w)=\prod_{i=1}^m\mathcal N(y_i\,|\,\mathbf w^\top\mathbf x_i,\sigma^2).
        \]
        ОМП по подвыборке $\mathfrak D_k\subseteq\mathfrak D_m$: $\hat{\mathbf w}_k=\argmax_{\mathbf w}L(\mathfrak D_k,\mathbf w)=(\mathbf X_k^\top\mathbf X_k)^{-1}\mathbf X_k^\top\mathbf y_k$.
    \end{block}
    \begin{block}{Апостериорное распределение}
        При $p(\mathbf w)=\mathcal N(\mathbf w\,|\,\mathbf m_0,\boldsymbol\Sigma_0)$:
        \ $p_k(\mathbf w)=p(\mathbf w\,|\,\mathfrak D_k)=\mathcal N(\mathbf w\,|\,\mathbf m_k,\boldsymbol\Sigma_k)$,
        \[
            \boldsymbol\Sigma_k^{-1}=\boldsymbol\Sigma_0^{-1}+\frac{1}{\sigma^2}\mathbf X_k^\top\mathbf X_k,
            \qquad
            \mathbf m_k=\boldsymbol\Sigma_k\Bigl(\boldsymbol\Sigma_0^{-1}\mathbf m_0+\frac{1}{\sigma^2}\mathbf X_k^\top\mathbf y_k\Bigr).
        \]
        \vspace{-1.3em}
    \end{block}
    \begin{block}{Теорема (Киселев, 2025)}
        Пусть $\|\mathbf x\|_2\leqslant M$, $|y|\leqslant M$ и~$\lambda_{\min}(\mathbf X_k^\top\mathbf X_k)=\omega(\sqrt k)$ при $k\to\infty$. Тогда в~линейной регрессии с~нормальным априорным распределением $\|\mathbf m_{k+1}-\mathbf m_k\|_2\to0$ и~$\|\boldsymbol\Sigma_{k+1}-\boldsymbol\Sigma_k\|_F\to0$.
    \end{block}
\end{frame}

\begin{frame}{Близость апостериорных распределений\chap{4}}

\begin{columns}[T]

\end{block}

\begin{frame}{Положения, выносимые на защиту}
    \begin{enumerate}
        \item Теоремы о~верхних оценках спектральной нормы гаусс"--~ньютоновской компоненты матрицы Гессе для полносвязных и~сверточных архитектур нейронных сетей.
        \item Критерии стабилизации ландшафта функции потерь: порядок $\mathcal O(k^{-1})$ и~$\mathcal O(k^{-2})$ следует из усреднения по объектам, а~оценки констант получены по локальной квадратичной аппроксимации в~одной точке, в~полном пространстве параметров и~в~подпространстве главных собственных направлений матрицы Гессе.

        \item Критерии достаточного размера выборки для параметрических моделей, основанные на стабилизации ландшафта функции потерь.
        \item Критерии достаточного размера выборки для вероятностных линейных моделей, основанные на стабилизации распределения оптимальных параметров при увеличении объема данных.
